cmake_minimum_required(VERSION 3.8)
project(robot_end_effector)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/pwm_motor_node.py scripts/serial_striker_node.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
install(FILES README.md SERIAL_STRIKER.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(pwm_motor_io test/test_pwm_motor_io.py TIMEOUT 30)
  ament_add_pytest_test(pwm_motor_action test/test_pwm_motor_action.py TIMEOUT 45)
  ament_add_pytest_test(encoder test/test_encoder.py TIMEOUT 30)
  ament_add_pytest_test(serial_striker_io test/test_serial_striker_io.py TIMEOUT 30)
  ament_add_pytest_test(serial_striker_action test/test_serial_striker_action.py TIMEOUT 45)
endif()
ament_package()
