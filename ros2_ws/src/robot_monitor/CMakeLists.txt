cmake_minimum_required(VERSION 3.8)
project(robot_monitor)
find_package(ament_cmake REQUIRED)
install(PROGRAMS scripts/robot_monitor.py DESTINATION lib/${PROJECT_NAME})
install(FILES scripts/monitor_core.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY web DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  enable_testing()
  add_test(NAME monitor_core COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/test -v)
endif()
ament_package()
